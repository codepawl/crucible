A scene in a few lines

Load a glTF model and an HDRI, then render.

import { Glaze } from 'crucible';
import { loadGltf } from 'crucible/addons/gltf';
import { loadHdri } from 'crucible/addons/hdri';
const { parts } = await loadGltf('/airport.glb');
const map = await loadHdri('/sky.hdr');
const renderer = new Glaze(canvas, parts, {
  hdr: true,
  ground: false,
  environment: { map },
});
renderer.render({ eye: [58, 31, 84], target: [-8, 3, -18] });
